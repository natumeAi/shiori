const _values = {
  'PreparedPublication',
  'ContentIdentity',
  'BookKey',
  'ServerKey',
  'PublicationSnapshot',
  'PublicationSectionSnapshot',
  'EpubPackage',
  'EpubManifestItem',
  'EpubSpineItem',
  'EpubTocItem',
  'PublicationMetadata',
  'PublicationLayoutFacts',
  'SectionLayoutFacts',
  // Audited final values and defensive unmodifiable copies; no lazy state.
  'SectionRendition',
  'RenditionIssue',
  'SectionSourceFacts',
  'SectionStyleFacts',
  'SvgSourceDimensions',
  'FirstSectionReadiness',
  'PublicationResourceReference',
};

/// Reflect every instance field, including private fields. A future accidental
/// archive/book/loader/callback field fails even if omitted by serialization.
int audit(Object? root) {
  final seen = HashSet<Object>.identity();
  var fields = 0;
  void visit(Object? value) {
    if (value == null ||
        value is String ||
        value is num ||
        value is bool ||
        value is Enum ||
        !seen.add(value)) {
      return;
    }
    if (value is Map) {
      for (final entry in value.entries) {
        visit(entry.key);
        visit(entry.value);
      }
      return;
    }
    if (value is Iterable) {
      for (final item in value) {
        visit(item);
      }
      return;
    }
    final instance = reflect(value);
    final name = MirrorSystem.getName(instance.type.simpleName);
    if (!_values.contains(name)) throw StateError('Non-value object: $name');
    for (
      ClassMirror? type = instance.type;
      type != null;
      type = type.superclass
    ) {
      for (final declaration in type.declarations.values) {
        if (declaration is! VariableMirror || declaration.isStatic) continue;
        if (!declaration.isFinal) throw StateError('Mutable field in $name');
        fields++;
        visit(instance.getField(declaration.simpleName).reflectee);
      }
    }
  }

  visit(root);
  return fields;
}

Future<void> main(List<String> args) async {
  final file = File(args.single);
  final key = BookKey(
    ServerKey(
      ServerEndpoint.parse('https://fixture.invalid/library').endpoint!,
    ),
    7,
  );
  final identity = ContentIdentity(
    key,
    crypto.sha256.convert(file.readAsBytesSync()).toString(),
  );
  final worker = await PublicationWorker.spawn();
  try {
    final result = await worker.prepare(file.path, identity);
    final fields = audit(result);
    // The ordinary publication has no rendition issues. Traverse an invalid
    // value too, so RenditionIssue's own final fields are actually inspected.
    final rawPackage = <String, String?>{'layout': 'future-layout'};
    final rawItemref = <String>['rendition:spread-diagonal'];
    final rendition = resolveSectionRendition(
      package: rawPackage,
      itemref: rawItemref,
    );
    if (audit(rendition) < 10 || rendition.issues.isEmpty) {
      throw StateError('Rendition audit did not visit issue values');
    }
    rawPackage.clear();
    rawItemref.clear();
    if (rendition.rawPackage.isEmpty || rendition.rawItemref.isEmpty) {
      throw StateError('Rendition retained mutable inputs');
    }
    for (final mutate in <void Function()>[
      () => rendition.rawPackage.clear(),
      () => rendition.rawItemref.clear(),
      () => rendition.effective.clear(),
      () => rendition.issues.clear(),
    ]) {
      var refused = false;
      try {
        mutate();
      } on UnsupportedError {
        refused = true;
      }
      if (!refused) throw StateError('Mutable rendition collection');
    }
    if (fields < 50) throw StateError('Audit did not visit the nested graph');
    final lazy = EpubBook.open(file.readAsBytesSync());
    for (final forbidden in <Object>[lazy, lazy.archive]) {
      var refused = false;
      try {
        audit(forbidden);
      } on StateError {
        refused = true;
      }
      if (!refused) throw StateError('Audit accepted a lazy object');
    }
    final archive = EpubArchive.openFile(file.path);
    archive.close();
    await worker.close();
    file.deleteSync();
    if (result.metadata.titles.single != 'Generated import') {
      throw StateError('Lost snapshot');
    }
    stdout.writeln(
      'PASS: $fields immutable fields audited; lazy book/archive refused; snapshot survives close/delete',
    );
  } finally {
    await worker.close();
  }
}
